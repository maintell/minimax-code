# PRC M1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a self-hosted Remote Control gateway plus web console that a desktop producer can join with a URL and one password, and drive a real session end to end.

**Architecture:** One standalone Node 24 process (`apps/prc`) owns an HTTP API, a producer WebSocket, a browser WebSocket, and a SQLite file. Protocol modules mirror the vendored bridge byte for byte so the shipped desktop client needs no protocol change. A small additive patch in the Electron main process turns it into the PRC credential source, keeping today's behaviour when PRC is unconfigured.

**Tech Stack:** Node 24 (native TypeScript type stripping, `node:sqlite`, `node:test`), the `ws` package, vanilla ES-module frontend, no bundler.

## Global Constraints

- Node `>= 24`: the service runs `.ts` sources directly; `tsc --noEmit` is the type gate.
- Runtime dependency budget: `ws` only. No framework, no ORM, no bundler, no native module.
- `apps/prc` is outside the pnpm workspace. Install with `pnpm install --ignore-workspace`, exactly like `apps/electron`.
- Protocol constants are copied verbatim and never invented: protocol version `1`, frame size floor `128 * 1024 * 1024`, `ctrl.ping` every `5000 ms`, producer idle timeout `15000 ms`, ticket TTL `60000 ms`, attachment caps `20 MB` per file and `50 MB` per batch.
- Every frame carries `v: 1`, `id`, `ts`, `type`, `payload`; outbound frames also carry a binding-scoped monotonic `event_id`.
- Producer-facing paths stay byte-compatible: base `/minimax-cloud/api/v1/remote-control`, routes `/pairing/code`, `/desktop/bindings`, `/unbind`.
- Unknown or malformed inbound frames are logged and dropped, never thrown.
- Passwords, tickets, signed URLs, message bodies and tool inputs never reach a log file.
- With no PRC configuration the desktop client behaves exactly as before; no new IPC channel, no new renderer surface.
- Documentation and commit messages are English.

```ts
// apps/prc/src/config.ts
import path from 'node:path';
import { fileURLToPath } from 'node:url';

export interface Config {
  port: number;
  dataDir: string;
  publicUrl: string;
  password?: string;
}

const appDir = path.dirname(path.dirname(fileURLToPath(import.meta.url)));

export function loadConfig(
  env: Record<string, string | undefined>,
  argv: string[],
): Config {
  const flagIndex = argv.indexOf('--password');
  const flagValue = flagIndex >= 0 ? argv[flagIndex + 1] : undefined;
  const password = flagValue ?? env.PRC_PASSWORD;
  const port = Number(env.PRC_PORT ?? 8787);
  if (!Number.isInteger(port) || port <= 0 || port > 65535) {
    throw new Error(`invalid PRC_PORT: ${String(env.PRC_PORT)}`);
  }
  const dataDir = path.resolve(env.PRC_DATA_DIR ?? path.join(appDir, 'data'));
  const publicUrl = (env.PRC_PUBLIC_URL ?? `http://127.0.0.1:${port}`).replace(/\/+$/u, '');
  return { port, dataDir, publicUrl, ...(password ? { password } : {}) };
}
```

```ts
// apps/prc/src/server.ts
import http from 'node:http';
import { readFile } from 'node:fs/promises';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import type { Config } from './config.ts';

const webDir = path.join(path.dirname(path.dirname(fileURLToPath(import.meta.url))), 'web');
const contentTypes: Record<string, string> = {
  '.html': 'text/html; charset=utf-8',
  '.js': 'text/javascript; charset=utf-8',
  '.css': 'text/css; charset=utf-8',
};

export async function createServer(deps: { config: Config }) {
  const server = http.createServer(async (req, res) => {
    const url = new URL(req.url ?? '/', 'http://prc.local');
    if (url.pathname === '/healthz') {
      res.writeHead(200, { 'content-type': 'application/json' });
      res.end(JSON.stringify({ ok: true }));
      return;
    }
    const file = url.pathname === '/' ? 'index.html' : url.pathname.slice(1);
    if (file.includes('..')) {
      res.writeHead(400).end();
      return;
    }
    try {
      const body = await readFile(path.join(webDir, file));
      res.writeHead(200, { 'content-type': contentTypes[path.extname(file)] ?? 'application/octet-stream' });
      res.end(body);
    } catch {
      res.writeHead(404).end();
    }
  });
  await new Promise<void>((resolve) => server.listen(deps.config.port, '127.0.0.1', resolve));
  const address = server.address();
  const port = typeof address === 'object' && address ? address.port : deps.config.port;
  return {
    url: `http://127.0.0.1:${port}`,
    close: () => new Promise<void>((resolve) => server.close(() => resolve())),
  };
}
```

```ts
// apps/prc/src/frames.ts
import { randomUUID } from 'node:crypto';

export const PROTOCOL_VERSION = 1;
export const DEFAULT_MAX_MESSAGE_BYTES = 128 * 1024 * 1024;
export const HEARTBEAT_INTERVAL_MS = 5_000;
export const INBOUND_IDLE_TIMEOUT_MS = 15_000;
export const TICKET_TTL_MS = 60_000;

export interface Frame {
  v: number;
  id: string;
  ts: number;
  type: string;
  payload: Record<string, unknown>;
  source?: string;
  in_reply_to?: string;
  trace_id?: string;
  event_id?: number;
}

export function makeFrame(
  type: string,
  payload: Record<string, unknown> = {},
  opts: { inReplyTo?: string; traceId?: string; eventId?: number } = {},
): Frame {
  return {
    v: PROTOCOL_VERSION,
    id: `f-${randomUUID()}`,
    ts: Date.now(),
    type,
    payload,
    source: 'server',
    ...(opts.inReplyTo ? { in_reply_to: opts.inReplyTo } : {}),
    ...(opts.traceId ? { trace_id: opts.traceId } : {}),
    ...(opts.eventId !== undefined ? { event_id: opts.eventId } : {}),
  };
}

function envelopeError(value: unknown): string | undefined {
  if (!value || typeof value !== 'object' || Array.isArray(value)) return 'frame is not an object';
  const frame = value as Record<string, unknown>;
  if (frame.v !== PROTOCOL_VERSION) return `unsupported protocol version: ${String(frame.v)}`;
  if (typeof frame.id !== 'string' || !frame.id) return 'missing id';
  if (typeof frame.type !== 'string' || !frame.type) return 'missing type';
  if (typeof frame.ts !== 'number' || !Number.isFinite(frame.ts)) return 'missing ts';
  if (frame.payload !== undefined &&
      (typeof frame.payload !== 'object' || frame.payload === null || Array.isArray(frame.payload))) {
    return 'payload must be an object';
  }
  return undefined;
}

export function decodeFrame(text: string, maxBytes = DEFAULT_MAX_MESSAGE_BYTES) {
  const size = Buffer.byteLength(text, 'utf8');
  if (size > maxBytes) {
    return { ok: false as const, error: { code: 'frame_too_large', message: `${size}B exceeds ${maxBytes}B` } };
  }
  let parsed: unknown;
  try {
    parsed = JSON.parse(text);
  } catch (error) {
    return { ok: false as const, error: { code: 'not_json', message: String(error) } };
  }
  const invalid = envelopeError(parsed);
  if (invalid) return { ok: false as const, error: { code: 'invalid_envelope', message: invalid } };
  const frame = parsed as Frame;
  return { ok: true as const, frame: { ...frame, payload: frame.payload ?? {} } };
}

export function ackTypeForCmd(type: string): string {
  return type.startsWith('cmd.') ? `ack.${type.slice(4)}` : 'ack.unknown';
}

export function errTypeForCmd(type: string): string {
  return type.startsWith('cmd.') ? `err.${type.slice(4)}` : 'err.unknown';
}
```

```ts
// apps/prc/src/auth.ts
import { createHmac, randomBytes, scryptSync, timingSafeEqual } from 'node:crypto';

const SCRYPT_N = 16_384;
const SCRYPT_R = 8;
const SCRYPT_P = 1;
const KEY_BYTES = 64;
const COOKIE_TTL_MS = 7 * 24 * 3600 * 1000;
const LOGIN_WINDOW_MS = 60_000;
const LOGIN_LIMIT = 5;

export function hashPassword(password: string): string {
  const salt = randomBytes(16);
  const key = scryptSync(password, salt, KEY_BYTES, { N: SCRYPT_N, r: SCRYPT_R, p: SCRYPT_P });
  return `scrypt$${SCRYPT_N}$${SCRYPT_R}$${SCRYPT_P}$${salt.toString('base64')}$${key.toString('base64')}`;
}

export function verifyPassword(password: string, stored: string): boolean {
  const parts = stored.split('$');
  if (parts.length !== 6 || parts[0] !== 'scrypt') return false;
  const [, n, r, p, salt, hash] = parts;
  const expected = Buffer.from(hash ?? '', 'base64');
  const actual = scryptSync(password, Buffer.from(salt ?? '', 'base64'), expected.length, {
    N: Number(n), r: Number(r), p: Number(p),
  });
  return expected.length === actual.length && timingSafeEqual(expected, actual);
}

function sign(secret: string, value: string): string {
  return createHmac('sha256', secret).update(value).digest('base64url');
}

export function createAuth(secret: string) {
  const attempts = new Map<string, number[]>();
  const makeToken = (payload: string) => `${Buffer.from(payload, 'utf8').toString('base64url')}.${sign(secret, payload)}`;

  return {
    issueCookie(nowMs = Date.now()): string {
      const expiresAt = nowMs + COOKIE_TTL_MS;
      return makeToken(`s|${expiresAt}`);
    },
    verifyCookie(raw: string | undefined, nowMs = Date.now()): boolean {
      if (!raw) return false;
      const [body, signature] = raw.split('.');
      if (!body || !signature) return false;
      const payload = Buffer.from(body, 'base64url').toString('utf8');
      const expected = sign(secret, payload);
      if (expected.length !== signature.length) return false;
      if (!timingSafeEqual(Buffer.from(expected), Buffer.from(signature))) return false;
      const [, expiresAt] = payload.split('|');
      return Number(expiresAt) > nowMs;
    },
    issueCsrf(): string {
      return randomBytes(24).toString('base64url');
    },
    verifyCsrf(cookie: string, header: string | undefined): boolean {
      if (!header || header.length !== cookie.length) return false;
      return timingSafeEqual(Buffer.from(cookie), Buffer.from(header));
    },
    allowLogin(key: string, nowMs = Date.now()): boolean {
      const fresh = (attempts.get(key) ?? []).filter((ts) => nowMs - ts < LOGIN_WINDOW_MS);
      if (fresh.length >= LOGIN_LIMIT) {
        attempts.set(key, fresh);
        return false;
      }
      fresh.push(nowMs);
      attempts.set(key, fresh);
      return true;
    },
  };
}

export type Auth = ReturnType<typeof createAuth>;
```

```ts
import { DatabaseSync } from 'node:sqlite';
import { TICKET_TTL_MS } from './frames.ts';

export interface Binding {
  id: number;
  deviceId: string;
  producerGeneration: number;
  producerVersion: string;
  boundAtMs: number;
  state: 'bound';
}

export function openStore(file: string) {
  const db = new DatabaseSync(file);
  db.exec(`
    PRAGMA journal_mode = WAL;
    CREATE TABLE IF NOT EXISTS binding (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      device_id TEXT NOT NULL UNIQUE,
      producer_generation INTEGER NOT NULL,
      producer_version TEXT NOT NULL,
      bound_at_ms INTEGER NOT NULL,
      state TEXT NOT NULL
    );
    CREATE TABLE IF NOT EXISTS ticket (
      hash TEXT PRIMARY KEY,
      binding_id INTEGER NOT NULL,
      issued_at_ms INTEGER NOT NULL,
      expires_at_ms INTEGER NOT NULL,
      consumed_at_ms INTEGER
    );
    CREATE TABLE IF NOT EXISTS event_log (
      binding_id INTEGER NOT NULL,
      event_id INTEGER NOT NULL,
      ts_ms INTEGER NOT NULL,
      frame_json TEXT NOT NULL,
      PRIMARY KEY (binding_id, event_id)
    );
    CREATE TABLE IF NOT EXISTS counter (
      binding_id INTEGER PRIMARY KEY,
      event_id INTEGER NOT NULL
    );
  `);

  const toBinding = (row: Record<string, unknown>): Binding => ({
    id: Number(row.id),
    deviceId: String(row.device_id),
    producerGeneration: Number(row.producer_generation),
    producerVersion: String(row.producer_version),
    boundAtMs: Number(row.bound_at_ms),
    state: 'bound',
  });

  return {
    getBinding(bindingId: number): Binding | undefined {
      const row = db.prepare('SELECT * FROM binding WHERE id = ?').get(bindingId);
      return row ? toBinding(row as Record<string, unknown>) : undefined;
    },
    findBindingByDevice(deviceId: string): Binding | undefined {
      const row = db.prepare('SELECT * FROM binding WHERE device_id = ?').get(deviceId);
      return row ? toBinding(row as Record<string, unknown>) : undefined;
    },
    getOrCreateBinding(deviceId: string, producerVersion: string): Binding {
      const existing = this.findBindingByDevice(deviceId);
      if (existing) {
        db.prepare('UPDATE binding SET producer_version = ? WHERE id = ?').run(producerVersion, existing.id);
        return { ...existing, producerVersion };
      }
      const max = db.prepare('SELECT COALESCE(MAX(producer_generation), 0) AS value FROM binding').get() as { value: number };
      const generation = Number(max.value) + 1;
      const result = db.prepare(
        'INSERT INTO binding (device_id, producer_generation, producer_version, bound_at_ms, state) VALUES (?, ?, ?, ?, ?)',
      ).run(deviceId, generation, producerVersion, Date.now(), 'bound');
      return this.getBinding(Number(result.lastInsertRowid)) as Binding;
    },
    deleteBinding(bindingId: number): void {
      db.prepare('DELETE FROM ticket WHERE binding_id = ?').run(bindingId);
      db.prepare('DELETE FROM event_log WHERE binding_id = ?').run(bindingId);
      db.prepare('DELETE FROM counter WHERE binding_id = ?').run(bindingId);
      db.prepare('DELETE FROM binding WHERE id = ?').run(bindingId);
    },
    mintTicket(bindingId: number, nowMs = Date.now()) {
      const ticket = randomBytes(32).toString('base64url');
      const hash = createHash('sha256').update(ticket).digest('hex');
      const expiresAtMs = nowMs + TICKET_TTL_MS;
      db.prepare('DELETE FROM ticket WHERE binding_id = ? AND consumed_at_ms IS NULL').run(bindingId);
      db.prepare('INSERT INTO ticket (hash, binding_id, issued_at_ms, expires_at_ms) VALUES (?, ?, ?, ?)')
        .run(hash, bindingId, nowMs, expiresAtMs);
      return { ticket, expiresAtMs };
    },
    consumeTicket(ticket: string, bindingId: number, nowMs = Date.now()): boolean {
      const hash = createHash('sha256').update(ticket).digest('hex');
      const row = db.prepare('SELECT * FROM ticket WHERE hash = ?').get(hash) as Record<string, unknown> | undefined;
      if (!row) return false;
      if (Number(row.binding_id) !== bindingId) return false;
      if (row.consumed_at_ms !== null) return false;
      if (Number(row.expires_at_ms) <= nowMs) return false;
      db.prepare('UPDATE ticket SET consumed_at_ms = ? WHERE hash = ?').run(nowMs, hash);
      return true;
    },
    nextEventId(bindingId: number): number {
      db.prepare('INSERT INTO counter (binding_id, event_id) VALUES (?, 0) ON CONFLICT(binding_id) DO NOTHING').run(bindingId);
      db.prepare('UPDATE counter SET event_id = event_id + 1 WHERE binding_id = ?').run(bindingId);
      const row = db.prepare('SELECT event_id FROM counter WHERE binding_id = ?').get(bindingId) as { event_id: number };
      return Number(row.event_id);
    },
    appendEvent(bindingId: number, eventId: number, frame: unknown): void {
      db.prepare('INSERT OR REPLACE INTO event_log (binding_id, event_id, ts_ms, frame_json) VALUES (?, ?, ?, ?)')
        .run(bindingId, eventId, Date.now(), JSON.stringify(frame));
      db.prepare(
        `DELETE FROM event_log WHERE binding_id = ? AND event_id <= (
           SELECT COALESCE(MAX(event_id), 0) - 2000 FROM event_log WHERE binding_id = ?)`,
      ).run(bindingId, bindingId);
    },
    replayAfter(bindingId: number, eventId: number): unknown[] {
      const rows = db.prepare('SELECT frame_json FROM event_log WHERE binding_id = ? AND event_id > ? ORDER BY event_id ASC')
        .all(bindingId, eventId) as Array<{ frame_json: string }>;
      return rows.map((row) => JSON.parse(row.frame_json));
    },
    close(): void {
      db.close();
    },
  };
}

export type Store = ReturnType<typeof openStore>;
```

```ts
// apps/prc/src/producer.ts
import { randomUUID } from 'node:crypto';
import { DEFAULT_MAX_MESSAGE_BYTES, decodeFrame, makeFrame, type Frame } from './frames.ts';
import type { Store } from './store.ts';

export interface WsLike {
  on(event: 'message', cb: (data: unknown) => void): void;
  on(event: 'close', cb: () => void): void;
  send(text: string): void;
  close(code?: number, reason?: string): void;
}

export interface Logger {
  info(message: string): void;
  warn(message: string): void;
  error(message: string): void;
}

export interface ProducerOptions {
  socket: WsLike;
  bindingId: number;
  store: Store;
  logger: Logger;
  onFrame(frame: Frame): void;
  onClose(): void;
}

export class ProducerSocket {
  readonly bindingId: number;
  private authed = false;
  private closed = false;
  private readonly connectionId = randomUUID();
  private maxBytes = DEFAULT_MAX_MESSAGE_BYTES;

  constructor(private readonly opts: ProducerOptions) {
    this.bindingId = opts.bindingId;
  }

  start(): void {
    this.sendFrame(makeFrame('ctrl.hello', {
      connection_id: this.connectionId,
      server_time_ms: Date.now(),
      max_message_bytes: this.maxBytes,
    }));
    this.opts.socket.on('message', (data) => this.handle(data));
    this.opts.socket.on('close', () => {
      this.closed = true;
      this.opts.onClose();
    });
  }

  isAuthed(): boolean {
    return this.authed;
  }

  send(type: string, payload: Record<string, unknown>, opts: { inReplyTo?: string; id?: string } = {}): Frame {
    const eventId = this.opts.store.nextEventId(this.bindingId);
    const frame = makeFrame(type, payload, { eventId, ...(opts.inReplyTo ? { inReplyTo: opts.inReplyTo } : {}) });
    if (opts.id) frame.id = opts.id;
    this.opts.store.appendEvent(this.bindingId, eventId, frame);
    this.sendFrame(frame);
    return frame;
  }

  close(reason: string): void {
    if (this.closed) return;
    this.sendFrame(makeFrame('ctrl.disconnect_reason', { code: reason }));
    this.closed = true;
    this.opts.socket.close(1000, reason);
  }

  private sendFrame(frame: Frame): void {
    try {
      this.opts.socket.send(JSON.stringify(frame));
    } catch (error) {
      this.opts.logger.warn(`producer send failed: ${String(error)}`);
    }
  }

  private handle(data: unknown): void {
    const text = typeof data === 'string' ? data : Buffer.isBuffer(data) ? data.toString('utf8') : String(data);
    const decoded = decodeFrame(text, this.maxBytes);
    if (!decoded.ok) {
      this.opts.logger.warn(`drop inbound frame: ${decoded.error.code} ${decoded.error.message}`);
      return;
    }
    const frame = decoded.frame;
    if (frame.type === 'ctrl.auth.complete') {
      this.authed = true;
      this.opts.logger.info(`producer authed binding=${this.bindingId} connection=${this.connectionId}`);
    }
    if (frame.type === 'ctrl.ping') {
      this.sendFrame(makeFrame('ctrl.pong', {}, { traceId: frame.trace_id }));
      return;
    }
    if (frame.type === 'ctrl.disconnect_reason') {
      this.opts.logger.info(`producer disconnect_reason ${String(frame.payload.code ?? 'unknown')}`);
    }
    this.opts.onFrame(frame);
  }
}
```

```ts
// apps/prc/src/commands.ts
import { randomUUID } from 'node:crypto';
import type { Frame } from './frames.ts';
import type { Logger, ProducerSocket } from './producer.ts';

export type CommandResult =
  | { kind: 'ack'; payload: Record<string, unknown> }
  | { kind: 'err'; code: string; message: string; retryable: boolean };

interface Pending {
  bindingId: number;
  resolve(result: CommandResult): void;
  timer: NodeJS.Timeout;
}

export class CommandRouter {
  private readonly pending = new Map<string, Pending>();
  private readonly lastByBinding = new Map<number, string>();

  constructor(private readonly deps: {
    getProducer(bindingId: number): ProducerSocket | undefined;
    logger: Logger;
    nowMs?: () => number;
  }) {}

  pendingCommandId(bindingId: number): string | undefined {
    return this.lastByBinding.get(bindingId);
  }

  request(
    bindingId: number,
    type: string,
    payload: Record<string, unknown>,
    opts: { timeoutMs?: number } = {},
  ): Promise<CommandResult> {
    const producer = this.deps.getProducer(bindingId);
    if (!producer || !producer.isAuthed()) {
      return Promise.resolve({ kind: 'err', code: 'PRODUCER_OFFLINE', message: 'producer is not connected', retryable: true });
    }
    const cmdId = `cmd-${randomUUID()}`;
    this.lastByBinding.set(bindingId, cmdId);
    return new Promise<CommandResult>((resolve) => {
      const timer = setTimeout(() => {
        this.pending.delete(cmdId);
        this.deps.logger.warn(`command timed out type=${type} id=${cmdId}`);
        resolve({ kind: 'err', code: 'COMMAND_TIMEOUT', message: `${type} timed out`, retryable: true });
      }, opts.timeoutMs ?? 30_000);
      this.pending.set(cmdId, { bindingId, resolve, timer });
      try {
        producer.send(type, payload, { id: cmdId });
        this.deps.logger.info(`command sent type=${type} id=${cmdId}`);
      } catch (error) {
        clearTimeout(timer);
        this.pending.delete(cmdId);
        resolve({ kind: 'err', code: 'PRODUCER_UNAVAILABLE', message: String(error), retryable: true });
      }
    });
  }

  settle(frame: Frame): boolean {
    const replyTo = frame.in_reply_to;
    if (!replyTo) return false;
    const pending = this.pending.get(replyTo);
    if (!pending) return false;
    this.pending.delete(replyTo);
    clearTimeout(pending.timer);
    if (frame.type.startsWith('ack.')) {
      pending.resolve({ kind: 'ack', payload: frame.payload });
      return true;
    }
    const code = typeof frame.payload.code === 'string' ? frame.payload.code : 'UNKNOWN';
    const message = typeof frame.payload.message === 'string' ? frame.payload.message : code;
    pending.resolve({ kind: 'err', code, message, retryable: frame.payload.retryable === true });
    return true;
  }

  cancelAll(bindingId: number, reason: string): void {
    for (const [id, pending] of this.pending) {
      if (pending.bindingId !== bindingId) continue;
      clearTimeout(pending.timer);
      pending.resolve({ kind: 'err', code: 'PRODUCER_DISCONNECTED', message: reason, retryable: true });
      this.pending.delete(id);
    }
  }
}
```

Note: the command frame is built by the socket so the id allocated here is the
id the producer echoes back in `in_reply_to`. Do not construct a second frame in
the router.

```ts
// apps/prc/src/http.ts
import type { IncomingMessage, ServerResponse } from 'node:http';
import { verifyPassword } from './auth.ts';
import type { Auth } from './auth.ts';
import type { CommandRouter } from './commands.ts';
import type { Config } from './config.ts';
import type { Store } from './store.ts';

const SESSION_COOKIE = 'prc_session';
const CSRF_COOKIE = 'prc_csrf';

function readCookie(req: IncomingMessage, name: string): string | undefined {
  const header = req.headers.cookie;
  if (!header) return undefined;
  for (const part of header.split(';')) {
    const [key, ...rest] = part.trim().split('=');
    if (key === name) return decodeURIComponent(rest.join('='));
  }
  return undefined;
}

async function readJson(req: IncomingMessage): Promise<Record<string, unknown>> {
  const chunks: Buffer[] = [];
  for await (const chunk of req) chunks.push(chunk as Buffer);
  if (!chunks.length) return {};
  return JSON.parse(Buffer.concat(chunks).toString('utf8')) as Record<string, unknown>;
}

function send(res: ServerResponse, status: number, body: unknown): void {
  res.writeHead(status, { 'content-type': 'application/json' });
  res.end(JSON.stringify(body));
}

export function createHttpApi(deps: {
  auth: Auth;
  store: Store;
  router: CommandRouter;
  config: Config;
  passwordHash: string;
  nowMs?: () => number;
}) {
  const now = deps.nowMs ?? Date.now;

  return {
    async handle(req: IncomingMessage, res: ServerResponse): Promise<boolean> {
      const url = new URL(req.url ?? '/', 'http://prc.local');
      if (!url.pathname.startsWith('/api/')) return false;

      if (req.method === 'POST' && url.pathname === '/api/login') {
        const key = req.socket.remoteAddress ?? 'unknown';
        if (!deps.auth.allowLogin(key, now())) return send(res, 429, { error: { code: 'RATE_LIMITED' } }), true;
        const body = await readJson(req);
        if (typeof body.password !== 'string' || !verifyPassword(body.password, deps.passwordHash)) {
          return send(res, 401, { error: { code: 'not_authenticated' } }), true;
        }
        const csrf = deps.auth.issueCsrf();
        res.setHeader('set-cookie', [
          `${SESSION_COOKIE}=${encodeURIComponent(deps.auth.issueCookie(now()))}; HttpOnly; SameSite=Lax; Path=/`,
          `${CSRF_COOKIE}=${csrf}; SameSite=Lax; Path=/`,
        ]);
        return send(res, 200, { ok: true, csrf }), true;
      }

      if (!deps.auth.verifyCookie(readCookie(req, SESSION_COOKIE), now())) {
        return send(res, 401, { error: { code: 'not_authenticated' } }), true;
      }

      const mutating = req.method !== 'GET' && req.method !== 'HEAD';
      if (mutating) {
        const csrf = readCookie(req, CSRF_COOKIE) ?? '';
        const header = req.headers['x-csrf-token'];
        if (!deps.auth.verifyCsrf(csrf, typeof header === 'string' ? header : undefined)) {
          return send(res, 403, { error: { code: 'csrf_failed' } }), true;
        }
      }

      const binding = deps.store.findBindingByDevice(String(req.headers['x-device-id'] ?? '')) ?? undefined;
      const bindingId = binding?.id;

      if (req.method === 'GET' && url.pathname === '/api/state') {
        return send(res, 200, {
          binding: binding ? { id: binding.id, deviceId: binding.deviceId, generation: binding.producerGeneration } : null,
        }), true;
      }

      if (req.method === 'GET' && url.pathname === '/api/sessions') {
        if (bindingId === undefined) return send(res, 409, { error: { code: 'NO_BINDING' } }), true;
        const result = await deps.router.request(bindingId, 'cmd.session.list', {
          ...(url.searchParams.get('agent') ? { agent: url.searchParams.get('agent') } : {}),
          ...(url.searchParams.get('keyword') ? { keyword: url.searchParams.get('keyword') } : {}),
        });
        return send(res, result.kind === 'ack' ? 200 : 502, result), true;
      }

      const historyMatch = /^\/api\/sessions\/([^/]+)\/messages$/u.exec(url.pathname);
      if (req.method === 'GET' && historyMatch) {
        if (bindingId === undefined) return send(res, 409, { error: { code: 'NO_BINDING' } }), true;
        const result = await deps.router.request(bindingId, 'cmd.session.history', {
          session_id: decodeURIComponent(historyMatch[1] ?? ''),
          ...(url.searchParams.get('before') ? { before_msg_id: url.searchParams.get('before') } : {}),
          ...(url.searchParams.get('limit') ? { limit: Number(url.searchParams.get('limit')) } : {}),
        });
        if (result.kind === 'err') return send(res, 502, result), true;
        const messages = JSON.parse(String(result.payload.messages_json ?? '[]')) as unknown;
        return send(res, 200, { messages, hasMore: result.payload.has_more === true, oldestMsgId: result.payload.oldest_msg_id ?? '' }), true;
      }

      return send(res, 404, { error: { code: 'not_found' } }), true;
    },
  };
}
```

```ts
// apps/prc/src/consumer.ts
import type { CommandRouter } from './commands.ts';
import type { WsLike } from './producer.ts';

export class ConsumerHub {
  private readonly byBinding = new Map<number, Set<WsLike>>();

  addSocket(socket: WsLike, bindingId: number): void {
    const set = this.byBinding.get(bindingId) ?? new Set<WsLike>();
    set.add(socket);
    this.byBinding.set(bindingId, set);
  }

  removeSocket(socket: WsLike, bindingId: number): void {
    this.byBinding.get(bindingId)?.delete(socket);
  }

  socketCount(bindingId: number): number {
    return this.byBinding.get(bindingId)?.size ?? 0;
  }

  broadcast(bindingId: number, message: unknown): void {
    const text = JSON.stringify(message);
    for (const socket of this.byBinding.get(bindingId) ?? []) {
      try {
        socket.send(text);
      } catch {
        // A closing socket is not an error for the caller.
      }
    }
  }
}

export async function handleConsumerFrame(
  hub: ConsumerHub,
  router: CommandRouter,
  socket: WsLike,
  bindingId: number,
  raw: string,
): Promise<void> {
  let message: Record<string, unknown>;
  try {
    message = JSON.parse(raw) as Record<string, unknown>;
  } catch {
    return;
  }
  const requestId = typeof message.requestId === 'string' ? message.requestId : undefined;
  if (message.kind === 'cmd' && typeof message.type === 'string') {
    const type = message.type.startsWith('cmd.') ? message.type : `cmd.${message.type}`;
    const payload = (message.payload ?? {}) as Record<string, unknown>;
    const result = await router.request(bindingId, type, payload);
    socket.send(JSON.stringify({ ...result, kind: result.kind, requestId }));
    return;
  }
  if (message.kind === 'permission' && typeof message.decision === 'string') {
    const requestIds = Array.isArray(message.requestIds) ? message.requestIds.map(String) : [];
    const decision = message.decision === 'deny' ? 2 : 1;
    const result = await router.request(bindingId, 'cmd.permission.respond', {
      request_ids: requestIds,
      decision,
    });
    socket.send(JSON.stringify({ ...result, requestId }));
    void hub;
  }
}
```

```js
let csrf = sessionStorage.getItem('prc.csrf') ?? '';
let bindingId = null;
let currentSession = null;
const $ = (id) => document.getElementById(id);

async function api(path, init = {}) {
  const headers = { 'content-type': 'application/json', ...(csrf ? { 'x-csrf-token': csrf } : {}) };
  const res = await fetch(path, { ...init, headers: { ...headers, ...(init.headers ?? {}) }, credentials: 'same-origin' });
  if (res.status === 401) throw new Error('unauthenticated');
  return res.json();
}

$('login').addEventListener('submit', async (event) => {
  event.preventDefault();
  const body = await api('/api/login', { method: 'POST', body: JSON.stringify({ password: $('password').value }) });
  csrf = body.csrf;
  sessionStorage.setItem('prc.csrf', csrf);
  $('login').hidden = true;
  $('console').hidden = false;
  await refresh();
});

async function refresh() {
  const state = await api('/api/state');
  bindingId = state.binding?.id ?? null;
  $('status').textContent = bindingId ? 'connected' : 'no producer';
  if (!bindingId) return;
  const list = await api('/api/sessions');
  renderSessions(list.payload?.sessions ?? []);
  if (!currentSession && list.payload?.sessions?.[0]) await openSession(list.payload.sessions[0].id);
}

function renderSessions(sessions) {
  $('sessions').replaceChildren(...sessions.map((session) => {
    const li = document.createElement('li');
    li.textContent = session.title || session.id;
    li.onclick = () => openSession(session.id);
    return li;
  }));
}

async function openSession(id) {
  currentSession = id;
  const page = await api(`/api/sessions/${encodeURIComponent(id)}/messages?limit=50`);
  $('transcript').replaceChildren();
  for (const message of page.messages ?? []) appendText(messageText(message));
  connectSocket();
}

function messageText(message) {
  return typeof message?.content === 'string' ? message.content : JSON.stringify(message);
}

function appendText(text) {
  const div = document.createElement('div');
  div.className = 'bubble';
  div.textContent = text;
  $('transcript').append(div);
}

$('composer').addEventListener('submit', async (event) => {
  event.preventDefault();
  const content = $('input').value.trim();
  if (!content || !currentSession) return;
  $('input').value = '';
  appendText(`me: ${content}`);
  await api(`/api/sessions/${encodeURIComponent(currentSession)}/messages`, {
    method: 'POST', body: JSON.stringify({ content }),
  });
});

$('stop').onclick = () => currentSession && api(`/api/sessions/${encodeURIComponent(currentSession)}/abort`, { method: 'POST' });

let socket;
function connectSocket() {
  socket?.close();
  socket = new WebSocket(`${location.protocol === 'https:' ? 'wss' : 'ws'}://${location.host}/ws`);
  socket.onmessage = (event) => {
    const message = JSON.parse(event.data);
    if (message.kind !== 'evt') return;
    if (message.type === 'evt.session.message_chunk' && message.payload?.session_id === currentSession) {
      appendText(message.payload.chunk?.text ?? '');
    }
    if (message.type === 'evt.session.turn_end') $('status').textContent = 'idle';
    if (message.type === 'evt.permission.ask') renderPermission(message.payload);
    if (message.type === 'evt.permission.resolved') {
      document.getElementById(`p-${message.payload?.request_id}`)?.remove();
    }
  };
}

function renderPermission(card) {
  const box = document.createElement('div');
  box.id = `p-${card.requestId}`;
  box.className = 'permission';
  box.innerHTML = `<strong>${card.toolName}</strong><pre>${card.toolInput}</pre>`;

        method: 'POST', body: JSON.stringify({ decision }),
      });
      box.remove();
    };
    box.append(button);
  }
  $('permissions').append(box);
```

No unit test exists for this task. `apps/electron` ships no test runner, and a
test file under `main/` would be compiled into the published client, so the gate
here is the type checker plus the manual smoke in task 14. Keep the module pure —
injected `fetchImpl`, no Electron import — so a harness can be added later
without refactoring.

```ts
// apps/electron/main/modules/remote-control-bridge/prc-credentials.ts
export interface PrcConfig {
  baseUrl: string;
  password: string;
  allowInsecureOrigin: boolean;
}

export interface PrcCredentials {
  binding_id: number;
  ticket: string;
  endpoint: string;
}

function text(value: unknown): string | undefined {
  return typeof value === 'string' && value.trim() ? value.trim() : undefined;
}

export function parsePrcConfig(raw: unknown): PrcConfig | undefined {
  if (!raw || typeof raw !== 'object' || Array.isArray(raw)) return undefined;
  const record = raw as Record<string, unknown>;
  const url = text(record.url);
  const password = text(record.password);
  if (!url || !password) return undefined;
  let parsed: URL;
  try {
    parsed = new URL(url);
  } catch {
    return undefined;
  }
  if (parsed.protocol !== 'https:' && parsed.protocol !== 'http:') return undefined;
  return {
    baseUrl: `${parsed.origin}${parsed.pathname.replace(/\/+$/u, '')}`,
    password,
    allowInsecureOrigin: record.allowInsecureOrigin === true,
  };
}

export function buildBindingUrl(baseUrl: string): string {
  return `${baseUrl}/minimax-cloud/api/v1/remote-control/desktop/bindings?rc_protocol_version=3`;
}

export async function fetchPrcBinding(deps: {
  config: PrcConfig;
  deviceId: string;
  clientVersion: string;
  fetchImpl: typeof fetch;
}): Promise<{ ok: true; credentials: PrcCredentials } | { ok: false; error: string }> {
  let response: Response;
  try {
    response = await deps.fetchImpl(buildBindingUrl(deps.config.baseUrl), {
      method: 'GET',
      headers: {
        'X-PRC-Password': deps.config.password,
        'X-Device-Id': deps.deviceId,
        'X-Producer-Type': 'desktop',
        'X-Client-Version': deps.clientVersion,
        Accept: 'application/json',
      },
    });
  } catch (error) {
    return { ok: false, error: `network:${error instanceof Error ? error.name : 'unknown'}` };
  }
  if (response.status === 401) return { ok: false, error: 'not_authenticated' };
  if (!response.ok) return { ok: false, error: `http_${response.status}` };
  let body: Record<string, unknown>;
  try {
    body = (await response.json()) as Record<string, unknown>;
  } catch {
    return { ok: false, error: 'not_bound' };
  }
  const bindingId = typeof body.binding_id === 'number' ? body.binding_id : undefined;
  const ticket = text(body.ticket);
  const endpoint = text(body.endpoint);
  if (body.state !== 'bound' || bindingId === undefined || !ticket || !endpoint) {
    return { ok: false, error: 'not_bound' };
  }
  return { ok: true, credentials: { binding_id: bindingId, ticket, endpoint } };
}
```

Rules encoded above: an empty URL or password disables the mode entirely; only
`http`/`https` are accepted; `state` must be `bound`; `binding_id` must be a
number; `ticket` and `endpoint` must be non-empty. No logging in this module —
callers own the audit trail and must never print the ticket.
